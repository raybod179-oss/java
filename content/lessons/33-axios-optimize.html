Axios

Axios کتابخانه‌ای است که کار با API را کوتاه‌تر می‌کند: خودش JSON را تبدیل می‌کند، نتیجه در response.data است و جواب ۴۰۴ و ۵۰۰ را مستقیم خطا می‌دهد (برخلاف fetch). برای استفاده فقط تگ آن را به HTML اضافه کن:

کدhtml
<script src="https://cdn.jsdelivr.net/npm/axios@1.7.9/dist/axios.min.js"></script>
<script>
const api = axios.create({ baseURL: "https://jsonplaceholder.typicode.com" });

async function load() {
  try {
    const { data } = await api.get("/posts", { params: { _limit: 3 } });
    console.log(data.map((p) => p.title));
    await api.post("/posts", { title: "جدید", userId: 1 });
    await api.delete("/posts/1");
  } catch (err) {
    if (err.response) console.log("خطای سرور:", err.response.status);
    else console.log("اتصال برقرار نشد");
  }
}
load();
</script>

Debounce: جلوگیری از درخواست‌های اضافه

در جستجوی زنده اگر با هر حرف یک درخواست بفرستی، سرور و کاربر هر دو اذیت می‌شوند. Debounce فقط وقتی کاربر چند لحظه تایپ نکرد، تابع را اجرا می‌کند.

کد · قابل اجراjs
function debounce(fn, ms) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), ms);
  };
}
const search = debounce((q) => console.log("جستجو برای:", q), 300);
search("j");
search("ja");
search("jav");          // فقط همین یکی اجرا می‌شود

کش ساده

کد · قابل اجراjs
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
const cache = new Map();
async function getUser(id) {
  if (cache.has(id)) { console.log("از حافظه"); return cache.get(id); }
  await wait(200);
  const user = { id };
  cache.set(id, user);
  console.log("از سرور");
  return user;
}
(async () => { await getUser(1); await getUser(1); })();

لغو درخواست با AbortController

وقتی کاربر صفحه را عوض می‌کند یا جستجوی جدیدی می‌زند، درخواست قبلی بی‌فایده است و باید لغو شود.

کد · قابل اجراjs
const ctrl = new AbortController();
ctrl.signal.addEventListener("abort", () => console.log("لغو شد"));
console.log(ctrl.signal.aborted);   // false
ctrl.abort();
console.log(ctrl.signal.aborted);   // true

// استفاده‌ی واقعی:
// fetch(url, { signal: ctrl.signal }).catch((e) => e.name === "AbortError" && console.log("لغو شد"));

تمرین ۱

debounce را با تأخیر ۵۰۰ میلی‌ثانیه برای تابع log بنویس و ۴ بار صدا بزن

دیدن پاسخ
کد · قابل اجراjs
const debounce = (fn, ms) => {
  let t;
  return (...a) => { clearTimeout(t); t = setTimeout(() => fn(...a), ms); };
};
const log = debounce((x) => console.log("آخرین مقدار:", x), 500);
["a", "b", "c", "d"].forEach(log);
تمرین ۲

چالش: تابعی بنویس که تا ۳ بار تلاش مجدد کند اگر درخواست شکست خورد

دیدن پاسخ
کد · قابل اجراjs
const wait = (ms) => new Promise((r) => setTimeout(r, ms));
let calls = 0;
const flaky = async () => { calls++; if (calls < 3) throw new Error("شکست " + calls); return "موفق"; };

async function retry(fn, times) {
  for (let i = 1; i <= times; i++) {
    try { return await fn(); }
    catch (e) { console.log("تلاش", i, "ناموفق:", e.message); await wait(100); }
  }
  throw new Error("همه‌ی تلاش‌ها ناموفق بود");
}
retry(flaky, 3).then((r) => console.log(r));
