پروژه‌ی موزیک پلیر

یک موزیک پلیر کامل: پخش و توقف، آهنگ قبلی و بعدی، نوار پیشرفت قابل کشیدن، کنترل صدا، لیست پخش و رفتن خودکار به آهنگ بعد. چون فایل موسیقی در دسترس نیست، سه ملودی کوتاه را با کد می‌سازیم (فایل WAV در حافظه)؛ جای آن‌ها می‌توانی آدرس mp3 های خودت را بگذاری. با تگ Audio و رویدادهای آن کار می‌کنی.

// ساخت یک فایل WAV ساده در حافظه از روی نت‌ها (موج سینوسی)
function makeWav(notes, noteSec = 0.5) {
  const n = Math.floor(RATE * noteSec), total = n * notes.length;
  const buf = new ArrayBuffer(44 + total * 2), v = new DataView(buf);
  const str = (o, s) => [...s].forEach((c, i) => v.setUint8(o + i, c.charCodeAt(0)));
  str(0, "RIFF"); v.setUint32(4, 36 + total * 2, true); str(8, "WAVE"); str(12, "fmt ");
  v.setUint32(16, 16, true); v.setUint16(20, 1, true); v.setUint16(22, 1, true);
  v.setUint32(24, RATE, true); v.setUint32(28, RATE * 2, true); v.setUint16(32, 2, true); v.setUint16(34, 16, true);
  str(36, "data"); v.setUint32(40, total * 2, true);
  notes.forEach((f, k) => {
    for (let i = 0; i < n; i++) {
      const env = Math.min(1, i / 300, (n - i) / 600);                 // نرم شدن ابتدا و انتهای هر نت
      v.setInt16(44 + (k * n + i) * 2, Math.sin(2 * Math.PI * f * i / RATE) * 0.35 * env * 32767, true);
    }
  });
  return URL.createObjectURL(new Blob([buf], { type: "audio/wav" }));
}
TRACKS.forEach((t) => (t.src = makeWav(t.notes)));

const audio = new Audio();
const $ = (id) => document.getElementById(id);
const fmt = (s) => Math.floor(s / 60) + ":" + String(Math.floor(s % 60)).padStart(2, "0");

function load(i, autoplay = false) {
  index = (i + TRACKS.length) % TRACKS.length;
  const t = TRACKS[index];
  audio.src = t.src;
  $("title").textContent = t.title;
  $("artist").textContent = t.artist;
  [...$("list").children].forEach((li, n) => li.classList.toggle("on", n === index));
  $("seek").value = 0; $("cur").textContent = "0:00";
  if (autoplay) audio.play();
}
function toggle() { audio.paused ? audio.play() : audio.pause(); }

  li.innerHTML = "<span></span><small></small>";
  li.firstChild.textContent = t.title;
  li.lastChild.textContent = t.artist;
  li.addEventListener("click", () => load(i, true));

قدم ۱: تگ Audio و رویدادهایش

جاوااسکریپت یک شیء Audio دارد که بدون هیچ تگی در HTML، فایل صوتی پخش می‌کند: play()، pause()، currentTime و volume. به‌جای نظرسنجی مداوم، به رویدادهایش گوش می‌دهیم: timeupdate هر چند لحظه خبر می‌دهد، ended وقتی آهنگ تمام شد و loadedmetadata وقتی طول آهنگ معلوم شد.

قدم ۲: نوار پیشرفت و صدا

نوار پیشرفت یک input type="range" از ۰ تا ۱۰۰ است. دو جهت کار می‌کند: با پخش، مقدارش از currentTime می‌آید؛ وقتی کاربر می‌کشد، رویداد input زمان را برعکس تنظیم می‌کند: currentTime = درصد × duration. صدا هم مستقیم audio.volume بین ۰ تا ۱ است.

JavaScriptjs
seek.addEventListener("input", (e) => {
  audio.currentTime = (e.target.value / 100) * audio.duration;
});

قدم ۳: لیست پخش و ساخت WAV

همه‌ی آهنگ‌ها در آرایه‌ی TRACKS هستند و تابع load(i) هر کدام را بارگیری می‌کند؛ با باقیمانده (%) از آخر به اول می‌رود. تابع makeWav نشان می‌دهد صدا در نهایت فقط عددهای پشت سرهم است: برای هر نمونه یک موج سینوسی با بسامد نت حساب می‌کنیم و در فرمت WAV می‌نویسیم. در پروژه‌ی واقعی، به‌جای آن فقط src: "music/song1.mp3" می‌گذاری.

اشتباه‌های رایج

پخش خودکار بدون اجازه

مرورگرها play() را قبل از اولین کلیک کاربر مسدود می‌کنند. همیشه پخش را از یک رویداد کلیک شروع کن.

NaN برای مدت آهنگ

تا loadedmetadata نیامده، audio.duration برابر NaN است؛ قبل از تقسیم آن را چک کن.

چالش‌ها

چالش ۱

حالت‌های تکرار (یک آهنگ، همه) و پخش تصادفی اضافه کن.

راهنمایی

یک متغیر mode و Math.random برای انتخاب آهنگ بعد.

چالش ۲

کلیدهای میان‌بر: Space برای پخش، ← → برای ۵ ثانیه جلو و عقب.

راهنمایی

روی keydown مقدار currentTime را تغییر بده.

چالش ۳

آخرین آهنگ و زمان پخش را در localStorage ذخیره کن.

راهنمایی

در timeupdate ذخیره کن و در شروع برنامه برگردان.

روش کار

هر چالش را روی نسخه‌ی خودت انجام بده. با اضافه کردن همین قابلیت‌ها، پروژه یک نمونه‌کار واقعی برای پورتفولیوی تو می‌شود.
