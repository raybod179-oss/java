<script>
// مقصد: شروع روز اول سال نو (نیمه‌شب به وقت تهران)
const TARGETS = {
  nowruz: { label: "۱ فروردین ۱۴۰۶", date: new Date("2027-03-21T00:00:00+03:30") },
  gregorian: { label: "۱ ژانویه ۲۰۲۷", date: new Date("2027-01-01T00:00:00+03:30") },
};

const select = document.getElementById("which");
const boxes = document.getElementById("boxes");
const done = document.getElementById("done");
const el = (id) => document.getElementById(id);

const fa = (n) => String(n).padStart(2, "0").replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[d]);

function tick() {
  const target = TARGETS[select.value];
  el("target").textContent = "تا " + target.label;
  const diff = target.date - Date.now();            // اختلاف به میلی‌ثانیه
  if (diff <= 0) {
    boxes.hidden = true;
    done.hidden = false;
    return;
  }
  boxes.hidden = false;
  done.hidden = true;
  const total = Math.floor(diff / 1000);
  el("d").textContent = fa(Math.floor(total / 86400));
  el("h").textContent = fa(Math.floor((total % 86400) / 3600));
  el("m").textContent = fa(Math.floor((total % 3600) / 60));
  el("s").textContent = fa(total % 60);
}

select.addEventListener("change", tick);
setInterval(tick, 1000);
tick();
</script>

قدم ۱: زمان مقصد

مقصد را با new Date("2027-03-21T00:00:00+03:30") می‌سازیم، یعنی شروع روز اول فروردین. بخش +03:30 منطقه‌ی زمانی ایران است تا نتیجه روی همه‌ی دستگاه‌ها یکسان باشد. دو مقصد را داخل یک آبجکت نگه می‌داریم و با select بینشان جابه‌جا می‌شویم.

JavaScriptjs
const target = new Date("2027-03-21T00:00:00+03:30");
const diff = target - Date.now();   // میلی‌ثانیه

قدم ۲: تبدیل به روز و ساعت

اختلاف به میلی‌ثانیه است. اول به ثانیه تبدیل می‌کنیم و بعد با تقسیم و باقیمانده (%) روز، ساعت و دقیقه را جدا می‌کنیم. اینجا عملگر باقیمانده که در درس عملگرها دیدی کاربرد واقعی پیدا می‌کند.

JavaScriptjs
const total = Math.floor(diff / 1000);
const days = Math.floor(total / 86400);
const hours = Math.floor((total % 86400) / 3600);
۵

قدم ۳: به‌روزرسانی هر ثانیه

setInterval(tick, 1000) تابع را هر ثانیه صدا می‌زند. حتماً یک‌بار هم مستقیم tick() را صدا می‌زنیم تا کاربر یک ثانیه صفحه‌ی خالی نبیند. تابع fa عدد را دو رقمی و فارسی می‌کند.

JavaScriptjs
setInterval(tick, 1000);
tick();

مفاهیمی که در این پروژه استفاده شد

این پروژه بخش‌هایی از درس‌های قبل را به هم وصل می‌کند. اگر جایی نفهمیدی، همان درس را دوباره بخوان:

  • Date و اختلاف تاریخ‌ها (درس ۱۵)
  • setInterval (درس ۲۶)
  • عملگر % و Math.floor (درس‌های ۴ و ۱۰)
  • Regex و replace برای اعداد فارسی (درس‌های ۹ و ۱۵)
۷

اشتباه‌های رایج

تبدیل نکردن منطقه‌ی زمانی

بدون +03:30، تاریخ بر اساس ساعت دستگاه کاربر حساب می‌شود و برای ایرانی‌های خارج کشور عدد متفاوت می‌شود.

اتکا به ساعت دستگاه

اگر ساعت گوشی غلط باشد، شمارش هم غلط است. برای اپ جدی، زمان را از سرور بگیر.
