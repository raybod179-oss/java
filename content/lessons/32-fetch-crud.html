POST، PUT، PATCH و DELETE

برای غیر از GET باید در گزینه‌های fetch سه چیز بدهی: method، headers (اعلام اینکه بدنه JSON است) و body که حتماً رشته است، پس با JSON.stringify می‌سازی‌اش.

توی این درس چه چیزهایی یاد می‌گیریم؟

  1. مقدمه
  2. تفاوت PUT و PATCH
  3. تابع کمکی تمیز برای همه‌ی متدها
  4. تمرین‌ها
۱

مقدمه

کد (HTML و JS) · قابل اجراhtml
<button data-m="POST">POST</button>
<button data-m="PUT">PUT</button>
<button data-m="PATCH">PATCH</button>
<button data-m="DELETE">DELETE</button>
<pre id="o" style="direction:ltr;white-space:pre-wrap"></pre>
<script>
const base = "https://jsonplaceholder.typicode.com/posts";
const json = { "Content-Type": "application/json" };
const calls = {
  POST: () => fetch(base, { method: "POST", headers: json,
    body: JSON.stringify({ title: "جدید", body: "متن", userId: 1 }) }),
  PUT: () => fetch(base + "/1", { method: "PUT", headers: json,
    body: JSON.stringify({ id: 1, title: "ویرایش کامل", body: "متن", userId: 1 }) }),
  PATCH: () => fetch(base + "/1", { method: "PATCH", headers: json,
    body: JSON.stringify({ title: "فقط عنوان" }) }),
  DELETE: () => fetch(base + "/1", { method: "DELETE" }),
};
document.querySelectorAll("button").forEach((b) => {
  b.onclick = async () => {
    const o = document.getElementById("o");
    o.textContent = "در حال ارسال...";
    try {
      const res = await calls[b.dataset.m]();
      const data = await res.json();
      o.textContent = res.status + " " + b.dataset.m + "\n" + JSON.stringify(data, null, 2);
    } catch (e) {
      o.textContent = "خطا: " + e.message;
    }
  };
});
</script>
نکته

این API فقط شبیه‌سازی می‌کند: پاسخ موفق می‌دهد ولی چیزی ذخیره نمی‌کند. در پروژه‌ی واقعی، سرور خودت داده را در دیتابیس می‌نویسد.

تفاوت PUT و PATCH

PUT کل رکورد را با نسخه‌ی جدید جایگزین می‌کند و باید همه‌ی فیلدها را بفرستی. PATCH فقط فیلدهایی را که فرستاده‌ای تغییر می‌دهد.

تابع کمکی تمیز برای همه‌ی متدها

کدjs
async function api(method, url, data) {
  const res = await fetch(url, {
    method,
    headers: { "Content-Type": "application/json" },
    body: data ? JSON.stringify(data) : undefined,
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.status === 204 ? null : res.json();
}

تمرین ۱

فرم ساده‌ای بساز که عنوان را با POST بفرستد و پاسخ را نشان دهد

دیدن پاسخ
کد (HTML و JS) · قابل اجراhtml
<input id="t" placeholder="عنوان">
<button id="b">ارسال</button>
<p id="o"></p>
<script>
document.getElementById("b").onclick = async () => {
  const o = document.getElementById("o");
  const title = document.getElementById("t").value.trim();
  if (!title) { o.textContent = "عنوان را بنویس"; return; }
  try {
    const res = await fetch("https://jsonplaceholder.typicode.com/posts", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ title, userId: 1 }),
    });
    const data = await res.json();
    o.textContent = "ساخته شد با شناسه " + data.id;
  } catch (e) { o.textContent = "خطا: " + e.message; }
};
</script>
