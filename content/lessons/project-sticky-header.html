<script>
const header = document.getElementById("top");
const links = [...document.querySelectorAll("#nav a")];
const sections = links.map((a) => document.querySelector(a.getAttribute("href")));

// ۱) سایه‌ی هدر بعد از کمی اسکرول
function onScroll() {
  header.classList.toggle("scrolled", window.scrollY > 10);
}
window.addEventListener("scroll", onScroll, { passive: true });
onScroll();

// ۲) پررنگ کردن لینکِ بخش دیده‌شده
function setActive(id) {
  links.forEach((a) => a.classList.toggle("active", a.getAttribute("href") === "#" + id));
}
if ("IntersectionObserver" in window) {
  const observer = new IntersectionObserver((entries) => {
    entries.forEach((entry) => {
      if (entry.isIntersecting) setActive(entry.target.id);
    });
  }, { rootMargin: "-40% 0px -55% 0px" });      // بخشی که وسط صفحه است
  sections.forEach((s) => observer.observe(s));
}
setActive("home");
</script>

قدم ۱: هدر چسبان با CSS

برای چسباندن هدر نیازی به جاوااسکریپت نیست: position: sticky; top: 0 کافی است. backdrop-filter پشت هدر را محو می‌کند و scroll-behavior: smooth روی html اسکرول نرم می‌سازد. scroll-padding-top هم جلوی رفتن عنوان زیر هدر را می‌گیرد.

CSScss
header { position: sticky; top: 0; }
html { scroll-behavior: smooth; scroll-padding-top: 64px; }
۴

قدم ۲: سایه بعد از اسکرول

با رویداد scroll بررسی می‌کنیم صفحه از بالا فاصله گرفته یا نه. گزینه‌ی { passive: true } به مرورگر می‌گوید این شنونده اسکرول را متوقف نمی‌کند، پس صفحه روان می‌ماند.

JavaScriptjs
window.addEventListener("scroll", () => {
  header.classList.toggle("scrolled", window.scrollY > 10);
}, { passive: true });
۵

قدم ۳: تشخیص بخش فعلی

به‌جای محاسبه‌ی فاصله‌ها با getBoundingClientRect در هر اسکرول، از IntersectionObserver استفاده می‌کنیم؛ مرورگر خودش خبر می‌دهد کدام بخش وارد ناحیه‌ی دید شده. با rootMargin فقط وسط صفحه را «ناحیه‌ی فعال» تعریف می‌کنیم.

JavaScriptjs
const observer = new IntersectionObserver((entries) => {
  entries.forEach((e) => e.isIntersecting && setActive(e.target.id));
}, { rootMargin: "-40% 0px -55% 0px" });

مفاهیمی که در این پروژه استفاده شد

این پروژه بخش‌هایی از درس‌های قبل را به هم وصل می‌کند. اگر جایی نفهمیدی، همان درس را دوباره بخوان:

  • رویدادها و classList.toggle (درس‌های ۲۲ و ۲۴)
  • آرایه و map (درس ۱۲)
  • شرط و عملگر in (درس‌های ۴ و ۵)
  • BOM: window.scrollY (درس ۲۷)

شنونده‌ی scroll سنگین

محاسبه‌ی سنگین داخل scroll صفحه را کند می‌کند. تا می‌توانی از IntersectionObserver یا CSS استفاده کن.

بدون scroll-padding-top، هنگام رفتن به یک بخش، عنوانش زیر هدر چسبان پنهان می‌شود.
