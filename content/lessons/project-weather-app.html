پروژه‌ی برنامه‌ی آب‌وهوا

async function getJSON(url) {
  const res = await fetch(url);
  if (!res.ok) throw new Error("خطای سرور " + res.status);
  return res.json();
}

async function loadWeather(name) {
  setState("در حال دریافت…");
  try {
    // درخواست ۱: تبدیل نام شهر به مختصات
    const geo = await getJSON("https://geocoding-api.open-meteo.com/v1/search?count=1&language=fa&name=" + encodeURIComponent(name));
    if (!geo.results || !geo.results.length) return setState("شهری با این نام پیدا نشد 😕", true);
    const { latitude, longitude, name: place, country } = geo.results[0];

    // درخواست ۲: آب‌وهوا با مختصات (وابسته به نتیجه‌ی اول)
    const w = await getJSON("https://api.open-meteo.com/v1/forecast?timezone=auto&forecast_days=5&latitude=" + latitude + "&longitude=" + longitude +
      "&current=temperature_2m,weather_code,wind_speed_10m,relative_humidity_2m&daily=weather_code,temperature_2m_max,temperature_2m_min");

    const [desc, emoji] = info(w.current.weather_code);
    $("place").textContent = place + (country ? "، " + country : "");
    $("em").textContent = emoji;
    $("temp").textContent = Math.round(w.current.temperature_2m) + "°";
    $("desc").textContent = desc;
    $("wind").textContent = "💨 " + Math.round(w.current.wind_speed_10m) + " km/h";
    $("hum").textContent = "💧 " + w.current.relative_humidity_2m + "٪";
    $("now").hidden = false;

    $("days").innerHTML = "";
    w.daily.time.forEach((date, i) => {
      const d = document.createElement("div");
      d.className = "day";
      d.innerHTML = "<small></small><div class='em'></div><b></b><small></small>";
      const [first, , , last] = d.children;
      first.textContent = new Date(date).toLocaleDateString("fa-IR", { weekday: "short" });
      d.children[1].textContent = info(w.daily.weather_code[i])[1];
      d.children[2].textContent = Math.round(w.daily.temperature_2m_max[i]) + "°";
      last.textContent = Math.round(w.daily.temperature_2m_min[i]) + "°";
      $("days").append(d);
    });
    setState("");
    try { localStorage.setItem("city", name); } catch (e) {}
  } catch (err) {
    setState("دریافت نشد: " + err.message, true);
  }
}

قدم ۱: دو درخواست وابسته

API آب‌وهوا مختصات می‌خواهد، نه نام شهر. پس اول از سرویس جغرافیایی (geocoding) مختصات را می‌گیریم و بعد با آن، آب‌وهوا را. چون درخواست دوم به نتیجه‌ی اولی وابسته است، با await پشت‌سرهم نوشته می‌شوند. اگر شهر پیدا نشد، همان‌جا با پیام روشن برمی‌گردیم.

JavaScriptjs
const geo = await getJSON(geoUrl);
const { latitude, longitude } = geo.results[0];
const w = await getJSON(forecastUrl(latitude, longitude));

قدم ۲: جدول کدها

API فقط یک عدد برای وضعیت هوا می‌دهد (مثلاً ۶۳). یک آبجکت جدول کد به توضیح و ایموجی نگه می‌داریم و با تابع info می‌خوانیم؛ اگر کد ناشناخته بود، مقدار پیش‌فرض برمی‌گرداند تا برنامه نشکند. این الگوی «آبجکت به‌جای if و else طولانی» را همیشه به کار ببر.

JavaScriptjs
const info = (code) => CODES[code] || ["نامشخص", "❔"];

قدم ۳: پیش‌بینی چندروزه

داده‌ی روزانه به‌صورت چند آرایه‌ی موازی می‌آید: time، temperature_2m_max و ... هرکدام ۵ عضو. با forEach روی تاریخ‌ها می‌چرخیم و از شماره‌ی i برای خواندن بقیه‌ی آرایه‌ها استفاده می‌کنیم. نام روز با toLocaleDateString("fa-IR") فارسی می‌شود.

مفاهیمی که در این پروژه استفاده شد

این پروژه بخش‌هایی از درس‌های قبل را به هم وصل می‌کند. اگر جایی نفهمیدی، همان درس را دوباره بخوان:

  • Async/Await و درخواست‌های وابسته (درس ۳۰)
  • Fetch و مدیریت خطا (درس ۳۱)
  • Destructuring آبجکت و آرایه (درس ۱۳)
  • Date و toLocaleDateString (درس ۱۵)
  • آبجکت به‌عنوان جدول و localStorage (درس‌های ۳ و ۲۷)
