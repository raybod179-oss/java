پروژه‌ی اسلایدر تصاویر

گالری اسلایدی تصاویر با تصویر بزرگ، تصاویر کوچک زیرش، پیمایش با کیبورد، نمایش تمام‌صفحه و بارگذاری تنبل (lazy). برخلاف اسلایدر نظرات، اینجا با تصویر کار می‌کنیم و مدیریت حالتِ «انتخاب‌شده» را یاد می‌گیری. تصاویر با SVG ساخته شده‌اند تا نیازی به فایل جدا نباشد.

<script>
// تصاویر به‌صورت SVG تولید می‌شوند تا نیازی به فایل نباشد؛ در پروژه‌ی واقعی آدرس عکس‌ها را بگذار.
const colors = [["#f59e0b", "#7c2d12"], ["#22c55e", "#064e3b"], ["#3b82f6", "#1e1b4b"], ["#ec4899", "#4a044e"], ["#14b8a6", "#134e4a"]];
const names = ["غروب", "جنگل", "اقیانوس", "گل‌ها", "کوهستان"];
const svg = (a, b, n) => "data:image/svg+xml;utf8," + encodeURIComponent(
  '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 320 200"><defs><linearGradient id="g" x1="0" y1="0" x2="1" y2="1">' +
  '<stop offset="0" stop-color="' + a + '"/><stop offset="1" stop-color="' + b + '"/></linearGradient></defs>' +
  '<rect width="320" height="200" fill="url(#g)"/><circle cx="' + (60 + n * 50) + '" cy="70" r="34" fill="rgba(255,255,255,.25)"/></svg>');
const photos = colors.map(([a, b], i) => ({ src: svg(a, b, i), title: names[i] }));

const main = document.getElementById("main");
const cap = document.getElementById("cap");
const thumbs = document.getElementById("thumbs");
const count = document.getElementById("count");
let index = 0;

photos.forEach((p, i) => {
  const b = document.createElement("button");
  b.setAttribute("aria-label", "تصویر " + (i + 1));
  const im = document.createElement("img");
  im.src = p.src; im.alt = p.title; im.loading = "lazy";     // بارگذاری تنبل
  b.append(im);
  b.addEventListener("click", () => show(i));
  thumbs.append(b);
});

function show(i) {
  index = (i + photos.length) % photos.length;
  main.src = photos[index].src;
  main.alt = photos[index].title;
  cap.textContent = photos[index].title;
  count.textContent = (index + 1).toLocaleString("fa-IR") + " از " + photos.length.toLocaleString("fa-IR");
  [...thumbs.children].forEach((b, n) => b.classList.toggle("on", n === index));
  main.style.animation = "none"; void main.offsetWidth; main.style.animation = "";   // شروع دوباره‌ی انیمیشن
}

document.getElementById("next").addEventListener("click", () => show(index + 1));
document.getElementById("prev").addEventListener("click", () => show(index - 1));
document.getElementById("gallery").addEventListener("keydown", (e) => {
  if (e.key === "ArrowLeft") show(index + 1);
  if (e.key === "ArrowRight") show(index - 1);
});
// پیش‌بارگذاری تصویر بعدی تا لغزش فوری باشد
photos.forEach((p) => { const im = new Image(); im.src = p.src; });
show(0);
</script>

پوشه‌ی پروژه و اجرا

برای ساخت روی کامپیوتر خودت یک پوشه با این ساختار بساز:

ساختار پوشهtxt
image-slider/
├── index.html   ← ساختار صفحه (HTML)
├── style.css    ← ظاهر (CSS)
└── script.js    ← منطق (JavaScript)
چطور اجرا کنم؟

فایل index.html را در مرورگر باز کن، یا در VS Code با افزونه‌ی Live Server روی آن راست‌کلیک و «Open with Live Server» را بزن تا با هر ذخیره، صفحه خودکار تازه شود.

قدم ۱: داده‌ی تصاویر

تصاویر را به‌جای HTML ثابت، در یک آرایه از آبجکت‌ها نگه می‌داریم: { src, title }. بقیه‌ی صفحه (تصویر بزرگ، تصاویر کوچک، شمارنده) از روی همین آرایه ساخته می‌شود. اضافه کردن یک تصویر یعنی فقط یک خط تازه در آرایه. در این نسخه‌ی آموزشی، تصاویر SVG تولیدشده‌اند تا فایل جدا لازم نباشد.

JavaScriptjs
const photos = [
  { src: "img/sunset.jpg", title: "غروب" },
  { src: "img/forest.jpg", title: "جنگل" },
];

قدم ۲: تابع show

مثل اسلایدر نظرات، یک متغیر index و یک تابع show همه‌چیز را کنترل می‌کنند: تصویر بزرگ، توضیح، شمارنده و تصویر کوچک فعال. این «یک منبع حقیقت» بودن، باگ‌ها را کم می‌کند. دو خط انیمیشن، با ترفند خواندن offsetWidth انیمیشن را دوباره شروع می‌کنند.

JavaScriptjs
function show(i) {
  index = (i + photos.length) % photos.length;
  main.src = photos[index].src;
}
۵

قدم ۳: عملکرد و دسترس‌پذیری

loading="lazy" باعث می‌شود تصاویر خارج از دید دیرتر بارگیری شوند. با ساختن new Image()، تصویرها را از قبل بارگیری می‌کنیم تا هنگام لغزش پرش نداشته باشیم. صفت alt همیشه باید پر باشد تا صفحه‌خوان‌ها و موتورهای جستجو تصویر را بفهمند.

JavaScriptjs
im.loading = "lazy";
im.alt = p.title;
