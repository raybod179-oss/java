<script>
const API = "https://jsonplaceholder.typicode.com/posts";
const PER_PAGE = 5;
const list = document.getElementById("list");
const state = document.getElementById("state");
const detail = document.getElementById("detail");
let page = 1;

function setState(text, isError = false, retry = false) {
  state.textContent = text;
  state.className = "state" + (isError ? " err" : "");
  if (retry) {
    const b = document.createElement("button");
    b.textContent = "تلاش دوباره";
    b.addEventListener("click", () => load());
    state.append(" ", b);
  }
}

async function getJSON(url) {
  const res = await fetch(url);
  if (!res.ok) throw new Error("خطای سرور " + res.status);
  return res.json();
}

async function load() {
  setState("در حال دریافت…");
  list.innerHTML = "";
  try {
    const posts = await getJSON(API + "?_page=" + page + "&_limit=" + PER_PAGE);
    posts.forEach((p) => {
      const card = document.createElement("div");
      card.className = "post";
      card.innerHTML = "<h3></h3><p></p>";
      card.querySelector("h3").textContent = p.title;
      card.querySelector("p").textContent = p.body.slice(0, 70) + "…";
      card.addEventListener("click", () => open(p.id));
      list.append(card);
    });
    setState("");
    document.getElementById("page").textContent = "صفحه " + page;
    document.getElementById("prev").disabled = page === 1;
  } catch (err) {
    setState("دریافت نشد: " + err.message, true, true);
  }
}

async function open(id) {
  detail.hidden = false;
  detail.textContent = "در حال دریافت…";
  try {
    const p = await getJSON(API + "/" + id);
    detail.textContent = "";
    const h = document.createElement("b"); h.textContent = p.title;
    const t = document.createElement("p"); t.textContent = p.body;
    detail.append(h, t);
  } catch (err) {
    detail.textContent = "خطا: " + err.message;
  }
}

document.getElementById("next").addEventListener("click", () => { page++; load(); });
document.getElementById("prev").addEventListener("click", () => { if (page > 1) { page--; load(); } });
load();
</script>

قدم ۱: تابع کمکی getJSON

کدِ تکراری «fetch، چک ok، json()» را یک‌بار در getJSON می‌نویسیم و در همه‌جا استفاده می‌کنیم. اگر روزی بخواهی هدر یا توکن اضافه کنی، فقط همین یک جا را عوض می‌کنی.

JavaScriptjs
async function getJSON(url) {
  const res = await fetch(url);
  if (!res.ok) throw new Error("خطای سرور " + res.status);
  return res.json();
}

قدم ۲: صفحه‌بندی با query string

بیشتر APIها صفحه‌بندی را با پارامتر در آدرس می‌پذیرند: ?_page=2&_limit=5. متغیر page را نگه می‌داریم، با دکمه‌ها زیاد و کم می‌کنیم و load() را دوباره صدا می‌زنیم. دکمه‌ی قبلی در صفحه‌ی اول disabled می‌شود.

JavaScriptjs
const posts = await getJSON(API + "?_page=" + page + "&_limit=" + PER_PAGE);
۵

قدم ۳: سه حالت: loading، خطا، موفق

هر درخواست واقعی سه وضعیت دارد و کاربر باید هر سه را ببیند: در حال دریافت، شکست (با دکمه‌ی تلاش دوباره) و موفق. تابع setState این پیام‌ها را یک‌جا مدیریت می‌کند. نبودن حالت خطا یکی از رایج‌ترین ضعف‌های پروژه‌های مبتدی است.
