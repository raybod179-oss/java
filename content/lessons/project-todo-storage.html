<script>
const KEY = "todos-v1";
const list = document.getElementById("list");
const empty = document.getElementById("empty");
let todos = load();                // [{ id, text, done }]
let filter = "all";

function load() {
  try { return JSON.parse(localStorage.getItem(KEY)) || []; } catch (e) { return []; }
}
function save() {
  try { localStorage.setItem(KEY, JSON.stringify(todos)); } catch (e) {}
}

function visible() {
  if (filter === "active") return todos.filter((t) => !t.done);
  if (filter === "done") return todos.filter((t) => t.done);
  return todos;
}

function render() {
  list.innerHTML = "";
  visible().forEach((t) => {
    const li = document.createElement("li");
    li.dataset.id = t.id;
    li.className = t.done ? "done" : "";
    li.innerHTML = '<input type="checkbox" aria-label="انجام شد"><span></span><button data-act="del">حذف</button>';
    li.querySelector("input").checked = t.done;
    li.querySelector("span").textContent = t.text;
    list.append(li);
  });
  empty.hidden = visible().length > 0;
  const left = todos.filter((t) => !t.done).length;
  document.getElementById("left").textContent = left.toLocaleString("fa-IR") + " کار باقی مانده";
  document.querySelectorAll("#filters button").forEach((b) => b.classList.toggle("on", b.dataset.f === filter));
}
function commit() { save(); render(); }          // هر تغییر: ذخیره و دوباره رسم

document.getElementById("add").addEventListener("submit", (e) => {
  e.preventDefault();
  const input = document.getElementById("text");
  const text = input.value.trim();
  if (!text) return;
  todos.push({ id: Date.now(), text, done: false });
  input.value = "";
  commit();
});

// Event Delegation: یک شنونده برای همه‌ی آیتم‌ها
list.addEventListener("click", (e) => {
  const li = e.target.closest("li");
  if (!li) return;
  const id = Number(li.dataset.id);
  if (e.target.dataset.act === "del") todos = todos.filter((t) => t.id !== id);
  else if (e.target.type === "checkbox") todos = todos.map((t) => (t.id === id ? { ...t, done: !t.done } : t));
  else return;
  commit();
});
list.addEventListener("dblclick", (e) => {                 // ویرایش با دابل‌کلیک
  const li = e.target.closest("li");
  if (!li || e.target.tagName !== "SPAN") return;
  const id = Number(li.dataset.id);
  const next = prompt("ویرایش:", e.target.textContent);
  if (next && next.trim()) { todos = todos.map((t) => (t.id === id ? { ...t, text: next.trim() } : t)); commit(); }
});
document.getElementById("filters").addEventListener("click", (e) => {
  if (e.target.dataset.f) { filter = e.target.dataset.f; render(); }
});
document.getElementById("clear").addEventListener("click", () => { todos = todos.filter((t) => !t.done); commit(); });
render();
</script>

۳

قدم ۱: شکل داده

هر کار یک آبجکت است: { id, text, done }. id را با Date.now() می‌سازیم تا برای هر کار یکتا باشد. همه‌ی کارها در آرایه‌ی todos هستند و بقیه‌ی برنامه فقط این آرایه را می‌خواند.

JavaScriptjs
todos.push({ id: Date.now(), text, done: false });
۴

قدم ۲: الگوی commit

هر تغییری (افزودن، حذف، تیک زدن) یک مسیر ثابت دارد: آرایه را عوض کن، سپس commit(). این تابع هم در localStorage ذخیره می‌کند و هم صفحه را دوباره می‌سازد. به‌جای push و ویرایش درجا، آرایه‌ی جدید با map و filter می‌سازیم (immutable) که خطاهای کمتری دارد.

JavaScriptjs
function commit() { save(); render(); }
todos = todos.map((t) => (t.id === id ? { ...t, done: !t.done } : t));
۵

قدم ۳: ذخیره‌سازی ایمن

localStorage فقط رشته نگه می‌دارد، پس با JSON.stringify می‌نویسیم و با JSON.parse می‌خوانیم. اگر داده‌ی خراب یا خالی بود، try/catch و || [] برنامه را از کار نمی‌اندازند. در پیش‌نمایش زنده‌ی همین صفحه ذخیره‌سازی بسته است، ولی در فایل دانلودی بعد از رفرش کارها می‌مانند.

JavaScriptjs
try { return JSON.parse(localStorage.getItem(KEY)) || []; } catch (e) { return []; }
۶

قدم ۴: Event Delegation

به‌جای وصل کردن شنونده به هر دکمه‌ی حذف، یک شنونده روی ul داریم و با e.target.closest("li") می‌فهمیم کدام آیتم است. آیتم‌هایی که بعداً ساخته می‌شوند هم خودکار کار می‌کنند.

JavaScriptjs
list.addEventListener("click", (e) => {
  const li = e.target.closest("li");
  if (!li) return;
});

اشتباه‌های رایج

ویرایش مستقیم آبجکت داخل آرایه

تغییر درجا ردیابی تغییرات را سخت می‌کند. آرایه یا آبجکت جدید بساز ({ ...t, done: true }).

ذخیره نکردن بعد از هر تغییر

اگر فقط گاهی ذخیره کنی، با بسته شدن تب کارها از دست می‌رود؛ برای همین همه‌چیز از commit() می‌گذرد.
