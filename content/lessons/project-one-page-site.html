// ۱) منوی موبایل
const links = document.getElementById("links");
const burger = document.getElementById("burger");
burger.addEventListener("click", () => {
  const open = links.classList.toggle("open");
  burger.setAttribute("aria-expanded", String(open));

links.addEventListener("click", () => links.classList.remove("open"));

// ۲) افکت تایپ
const words = ["سایت می‌سازیم", "فروشگاه می‌سازیم", "ایده را اجرا می‌کنیم"];
const typed = document.getElementById("typed");
let w = 0, c = 0, deleting = false;
function type() {
  const word = words[w];
  c += deleting ? -1 : 1;
  typed.textContent = word.slice(0, c);
  let delay = deleting ? 40 : 90;
  if (!deleting && c === word.length) { deleting = true; delay = 1200; }
  else if (deleting && c === 0) { deleting = false; w = (w + 1) % words.length; delay = 300; }
  setTimeout(type, delay);
}
type();

// ۳) ظاهر شدن کارت‌ها و شمارنده با اسکرول
function countUp(el) {
  const to = Number(el.dataset.to);
  let n = 0;
  const step = () => { n += Math.ceil(to / 40); el.textContent = Math.min(n, to); if (n < to) requestAnimationFrame(step); };
  step();
}
if ("IntersectionObserver" in window) {
  const io = new IntersectionObserver((entries) => {
    entries.forEach((e) => {
      if (!e.isIntersecting) return;
      e.target.classList.add("in");
      if (e.target.dataset.to) countUp(e.target);
      io.unobserve(e.target);                        // فقط یک بار
    });
  }, { threshold: 0.3 });
  document.querySelectorAll(".card, [data-to]").forEach((el) => io.observe(el));
} else {
  document.querySelectorAll(".card").forEach((el) => el.classList.add("in"));
}

// ۴) فرم تماس
document.getElementById("form").addEventListener("submit", (e) => {
  e.preventDefault();
  const err = document.getElementById("err");
  const name = document.getElementById("name").value.trim();
  const email = document.getElementById("email").value.trim();
  const msg = document.getElementById("msg").value.trim();
  err.className = "err";
  if (name.length < 2) return (err.textContent = "نام را وارد کن");
  if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return (err.textContent = "ایمیل معتبر نیست");
  if (msg.length < 10) return (err.textContent = "پیام حداقل ۱۰ کاراکتر باشد");
  err.className = "err ok";
  err.textContent = "پیام شما ثبت شد ✓";
  e.target.reset();
});

پوشه‌ی پروژه و اجرا

برای ساخت روی کامپیوتر خودت یک پوشه با این ساختار بساز:

ساختار پوشهtxt
one-page-site/
├── index.html   ← ساختار صفحه (HTML)
├── style.css    ← ظاهر (CSS)
└── script.js    ← منطق (JavaScript)
چطور اجرا کنم؟

فایل index.html را در مرورگر باز کن، یا در VS Code با افزونه‌ی Live Server روی آن راست‌کلیک و «Open with Live Server» را بزن تا با هر ذخیره، صفحه خودکار تازه شود.

قدم ۱: ساختار و منوی موبایل

سایت از چند section با شناسه ساخته شده و لینک‌های منو با href="#services" به آن‌ها می‌روند؛ با scroll-behavior: smooth نرم می‌روند. در صفحه‌ی کوچک، لینک‌ها پنهان‌اند و دکمه‌ی همبرگری کلاس open را toggle می‌کند. صفت aria-expanded وضعیت منو را به صفحه‌خوان می‌گوید.

JavaScriptjs
burger.addEventListener("click", () => {
  const open = links.classList.toggle("open");
  burger.setAttribute("aria-expanded", String(open));
});

قدم ۲: افکت تایپ بدون کتابخانه

یک تابع که با setTimeout خودش را دوباره صدا می‌زند: هر بار یک حرف اضافه یا کم می‌کند و بسته به وضعیت (deleting) تأخیر را عوض می‌کند. کتابخانه‌ی Typed.js در بخش پلاگین‌ها همین کار را با یک خط انجام می‌دهد، ولی اینجا می‌فهمی چطور کار می‌کند.

JavaScriptjs
typed.textContent = word.slice(0, c);
setTimeout(type, deleting ? 40 : 90);
۵

قدم ۳: انیمیشن با اسکرول

کارت‌ها ابتدا شفاف و پایین‌ترند. با IntersectionObserver وقتی وارد دید شدند، کلاس in می‌گیرند و CSS با transition آن‌ها را بالا می‌آورد. شمارنده‌ها هم از requestAnimationFrame استفاده می‌کنند. بعد از اولین نمایش unobserve می‌زنیم تا انیمیشن تکرار نشود.

JavaScriptjs
io.observe(card);
// در callback: e.target.classList.add("in"); io.unobserve(e.target);

مفاهیمی که در این پروژه استفاده شد

این پروژه بخش‌هایی از درس‌های قبل را به هم وصل می‌کند. اگر جایی نفهمیدی، همان درس را دوباره بخوان:

  • DOM، classList و رویدادها (درس‌های ۲۲ و ۲۴)
  • setTimeout بازگشتی و requestAnimationFrame (درس ۲۶)
  • Regex و اعتبارسنجی فرم (درس ۱۵)
  • IntersectionObserver و scroll (درس‌های ۲۷ و ۳۶)
  • رشته: slice و trim (درس ۹)
