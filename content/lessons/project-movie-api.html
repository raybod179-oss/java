پروژه‌ی Movie API

جستجوی فیلم و سریال با یک API واقعی: تایپ می‌کنی، نتیجه‌ها با پوستر، امتیاز و سال می‌آیند و با کلیک خلاصه‌ی داستان را می‌بینی. این پروژه چهار مهارت حرفه‌ای را تمرین می‌کند: debounce، لغو درخواست قبلی با AbortController، حالت‌های loading و خطا و تمیز کردن متن HTML دریافتی. <div class="app">
  <h1>جستجوی فیلم و سریال</h1>
  <input id="q" type="search" placeholder="مثلاً: batman یا friends" aria-label="جستجو" autocomplete="off">
  <p class="state" id="state">نام یک فیلم یا سریال را (انگلیسی) بنویس</p>

<script>
const API = "https://api.tvmaze.com/search/shows?q=";
const grid = document.getElementById("grid");
const state = document.getElementById("state");
const detail = document.getElementById("detail");
let controller = null;            // برای لغو درخواست قبلی
let timer = null;

function setState(text, error = false) {
  state.textContent = text;
  state.className = "state" + (error ? " err" : "");
}
// متن خلاصه‌ی داستان HTML دارد؛ فقط متن خالصش را می‌گیریم
const plain = (html) => new DOMParser().parseFromString(html || "", "text/html").body.textContent;

async function search(text) {
  if (controller) controller.abort();               // درخواست قبلی بی‌فایده است
  controller = new AbortController();
  grid.innerHTML = ""; detail.hidden = true;
  if (!text) return setState("نام یک فیلم یا سریال را (انگلیسی) بنویس");
  setState("در حال جستجو…");
  try {
    const res = await fetch(API + encodeURIComponent(text), { signal: controller.signal });
    if (!res.ok) throw new Error("خطای سرور " + res.status);
    const results = await res.json();
    if (!results.length) return setState("چیزی پیدا نشد 😕");
    setState("");
    results.slice(0, 12).forEach(({ show }) => grid.append(card(show)));
  } catch (err) {
    if (err.name === "AbortError") return;           // لغو عمدی، خطا نیست
    setState("خطا: " + err.message, true);
  }
}

function card(show) {
  const c = document.createElement("div");
  c.className = "card";
  const img = document.createElement("img");
  img.alt = show.name; img.loading = "lazy";
  img.src = show.image ? show.image.medium : "data:image/svg+xml;utf8,<svg xmlns='http://www.w3.org/2000/svg' width='2' height='3'/>";
  const info = document.createElement("div");
  const year = show.premiered ? show.premiered.slice(0, 4) : "—";
  const rate = show.rating && show.rating.average ? "★ " + show.rating.average : "";
  info.innerHTML = "<b></b><br><small></small> <span class='rate'></span>";
  info.querySelector("b").textContent = show.name;
  info.querySelector("small").textContent = year;
  info.querySelector(".rate").textContent = rate;
  c.append(img, info);
  c.addEventListener("click", () => {
    detail.hidden = false;
    detail.innerHTML = "<h2></h2><p></p>";
    detail.querySelector("h2").textContent = show.name + " (" + year + ")";
    detail.querySelector("p").textContent = plain(show.summary) || "خلاصه‌ای ثبت نشده.";
  });
  return c;
}

document.getElementById("q").addEventListener("input", (e) => {
  clearTimeout(timer);                               // debounce: ۴۰۰ میلی‌ثانیه بعد از آخرین حرف
  timer = setTimeout(() => search(e.target.value.trim()), 400);
});
</script>

قدم ۱: Debounce

اگر با هر حرف تایپ‌شده یک درخواست بفرستیم، برای «batman» نه درخواست می‌رود و سرور و کاربر هر دو اذیت می‌شوند. با setTimeout و clearTimeout فقط وقتی کاربر ۴۰۰ میلی‌ثانیه تایپ نکرد جستجو می‌کنیم. همان الگوی درس ۳۳.

clearTimeout(timer);

قدم ۲: لغو درخواست قبلی

ممکن است جواب «bat» دیرتر از «batman» برسد و نتیجه‌ی اشتباه را نشان بدهد (به این باگ race condition می‌گویند). قبل از هر جستجو، درخواست قبلی را با AbortController لغو می‌کنیم. لغو یک AbortError پرتاب می‌کند که خطای واقعی نیست و بی‌صدا نادیده‌اش می‌گیریم.

JavaScriptjs
if (controller) controller.abort();
controller = new AbortController();
fetch(url, { signal: controller.signal });

قدم ۳: چهار حالت صفحه و امنیت

صفحه چهار حالت دارد: خالی، در حال جستجو، بدون نتیجه، و خطا؛ هرکدام پیام جدا دارند. خلاصه‌ی داستان از سرور با تگ HTML می‌آید. به‌جای innerHTML، آن را با DOMParser به متن خالص تبدیل و با textContent نمایش می‌دهیم تا کد مخرب اجرا نشود. تصویر هم loading="lazy" دارد.

  new DOMParser().parseFromString(html || "", "text/html").body.textContent;

مفاهیمی که در این پروژه استفاده شد

این پروژه بخش‌هایی از درس‌های قبل را به هم وصل می‌کند. اگر جایی نفهمیدی، همان درس را دوباره بخوان:

  • Fetch، res.ok و مدیریت خطا (درس‌های ۳۱ و ۳۰)

  • Destructuring در پارامتر ({ show }) (درس ۱۳)
  • encodeURIComponent و رشته‌ها (درس ۹)
  • ساخت المنت امن (درس ۲۳)

اشتباه‌های رایج

فراموش کردن encodeURIComponent

فاصله و حروف خاص در آدرس را خراب می‌کنند. ورودی کاربر را همیشه encode کن.

کلید API داخل کد سمت کاربر

APIهای کلیددار (مثل TMDB) را از طریق سرور خودت صدا بزن؛ هر کلیدی در مرورگر دیده می‌شود.
